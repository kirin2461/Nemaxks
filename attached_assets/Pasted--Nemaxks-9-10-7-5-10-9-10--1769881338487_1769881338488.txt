// backend/middleware_csrf.go
func CSRFMiddleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        if c.Request.Method != "GET" && c.Request.Method != "HEAD" {
            token := c.GetHeader("X-CSRF-Token")
            sessionToken := getCSRFFromSession(c)
            if token == "" || token != sessionToken {
                c.AbortWithStatusJSON(403, gin.H{"error": "CSRF token mismatch"})
                return
            }
        }

    }
}

### 3.2 DDoS Protection Layer
```go
// backend/middleware_ddos.go
type DDoSProtection struct {
    // IP reputation tracking
    ipScores map[string]float64
    
    // Behavioral analysis
    requestPatterns map[string][]RequestPattern
    
    // Auto-ban thresholds
    banThreshold    float64
    challengeThreshold float64
}

func (d *DDoSProtection) Middleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        ip := c.ClientIP()
        score := d.calculateThreatScore(ip, c.Request)
        
        if score > d.banThreshold {
            d.banIP(ip, 1*time.Hour)
            c.AbortWithStatus(403)
            return
        }
        
        if score > d.challengeThreshold {
            // Требовать CAPTCHA
            c.JSON(429, gin.H{"challenge": "captcha_required"})
            return
        }
        
        c.Next()
    }
}
```

    # SQL Injection
